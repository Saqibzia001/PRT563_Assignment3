// A. CENTRALITY - FINAL OPERATIONAL GRAPH
CALL gds.graph.drop('operationalCentrality', false);

MATCH (source)
WHERE source:Customer OR source:Address OR source:Store OR source:Driver
   OR source:MenuCategory OR source:MenuItem OR source:MenuVariant
   OR source:Order OR source:Payment OR source:Delivery
OPTIONAL MATCH (source)-[r]->(target)
WHERE target IS NULL
   OR target:Customer OR target:Address OR target:Store OR target:Driver
   OR target:MenuCategory OR target:MenuItem OR target:MenuVariant
   OR target:Order OR target:Payment OR target:Delivery
WITH gds.graph.project(
    'operationalCentrality',
    source,
    target,
    {
        sourceNodeLabels: labels(source),
        targetNodeLabels: CASE WHEN target IS NULL THEN [] ELSE labels(target) END,
        relationshipType: CASE WHEN r IS NULL THEN null ELSE type(r) END
    },
    {undirectedRelationshipTypes: ['*']}
) AS g
RETURN g.graphName AS graph, g.nodeCount AS nodes, g.relationshipCount AS relationships;

CALL gds.degree.stream('operationalCentrality')
YIELD nodeId, score
RETURN labels(gds.util.asNode(nodeId)) AS Labels,
       coalesce(gds.util.asNode(nodeId).orderId,
                gds.util.asNode(nodeId).menuItemId,
                gds.util.asNode(nodeId).categoryId,
                gds.util.asNode(nodeId).customerId,
                gds.util.asNode(nodeId).storeId,
                gds.util.asNode(nodeId).driverId,
                gds.util.asNode(nodeId).addressId,
                gds.util.asNode(nodeId).paymentId,
                gds.util.asNode(nodeId).deliveryId) AS BusinessID,
       score
ORDER BY score DESC
LIMIT 15;

CALL gds.betweenness.stream('operationalCentrality')
YIELD nodeId, score
RETURN labels(gds.util.asNode(nodeId)) AS Labels,
       coalesce(gds.util.asNode(nodeId).orderId,
                gds.util.asNode(nodeId).menuItemId,
                gds.util.asNode(nodeId).categoryId,
                gds.util.asNode(nodeId).customerId,
                gds.util.asNode(nodeId).storeId,
                gds.util.asNode(nodeId).driverId,
                gds.util.asNode(nodeId).addressId,
                gds.util.asNode(nodeId).paymentId,
                gds.util.asNode(nodeId).deliveryId) AS BusinessID,
       score
ORDER BY score DESC
LIMIT 15;

CALL gds.closeness.stream('operationalCentrality')
YIELD nodeId, score
RETURN labels(gds.util.asNode(nodeId)) AS Labels,
       coalesce(gds.util.asNode(nodeId).orderId,
                gds.util.asNode(nodeId).menuItemId,
                gds.util.asNode(nodeId).categoryId,
                gds.util.asNode(nodeId).customerId,
                gds.util.asNode(nodeId).storeId,
                gds.util.asNode(nodeId).driverId,
                gds.util.asNode(nodeId).addressId,
                gds.util.asNode(nodeId).paymentId,
                gds.util.asNode(nodeId).deliveryId) AS BusinessID,
       score
ORDER BY score DESC
LIMIT 15;

CALL gds.pageRank.stream('operationalCentrality', {dampingFactor: 0.85})
YIELD nodeId, score
RETURN labels(gds.util.asNode(nodeId)) AS Labels,
       coalesce(gds.util.asNode(nodeId).orderId,
                gds.util.asNode(nodeId).menuItemId,
                gds.util.asNode(nodeId).categoryId,
                gds.util.asNode(nodeId).customerId,
                gds.util.asNode(nodeId).storeId,
                gds.util.asNode(nodeId).driverId,
                gds.util.asNode(nodeId).addressId,
                gds.util.asNode(nodeId).paymentId,
                gds.util.asNode(nodeId).deliveryId) AS BusinessID,
       score
ORDER BY score DESC
LIMIT 15;

CALL gds.graph.drop('operationalCentrality', false);

// B. NODE SIMILARITY - JACCARD
CALL gds.graph.drop('customerVariantSimilarity', false);

MATCH (source:Customer)-[:PLACES]->(o:Order)-[:CONTAINS]->(target:MenuVariant)
WITH gds.graph.project(
    'customerVariantSimilarity',
    source,
    target,
    {relationshipType: 'PURCHASED_VARIANT'}
) AS g
RETURN g.graphName AS graph, g.nodeCount AS nodes, g.relationshipCount AS relationships;

CALL gds.nodeSimilarity.stream('customerVariantSimilarity', {
    topK: 3,
    similarityMetric: 'JACCARD'
})
YIELD node1, node2, similarity
WITH gds.util.asNode(node1) AS n1,
     gds.util.asNode(node2) AS n2,
     similarity
WHERE n1:Customer AND n2:Customer
RETURN n1.customerId AS Customer1,
       n2.customerId AS Customer2,
       similarity
ORDER BY similarity DESC, Customer1, Customer2;

CALL gds.graph.drop('customerVariantSimilarity', false);

// C. FILTERED NODE SIMILARITY - OVERLAP
MATCH (c:Customer)-[:PLACES]->(:Order)-[:CONTAINS]->(:MenuVariant)
      <-[:HAS_VARIANT]-(m:MenuItem)<-[:CLASSIFIES]-(cat:MenuCategory)
WITH DISTINCT c, 'CATEGORY:' + toString(cat.categoryId) AS key,
     'Category' AS featureType, cat.categoryName AS featureValue
MERGE (f:AffinityFeature {featureKey: key})
SET f.featureType = featureType, f.featureValue = featureValue
MERGE (c)-[:HAS_AFFINITY_FEATURE]->(f);

MATCH (c:Customer)-[:PLACES]->(o:Order)
WITH DISTINCT c, 'STATUS:' + o.orderStatus AS key,
     'OrderStatus' AS featureType, o.orderStatus AS featureValue
MERGE (f:AffinityFeature {featureKey: key})
SET f.featureType = featureType, f.featureValue = featureValue
MERGE (c)-[:HAS_AFFINITY_FEATURE]->(f);

CALL gds.graph.drop('customerAffinitySimilarity', false);
MATCH (source:Customer)-[:HAS_AFFINITY_FEATURE]->(target:AffinityFeature)
WITH gds.graph.project(
    'customerAffinitySimilarity',
    source,
    target,
    {
        sourceNodeLabels: 'Customer',
        targetNodeLabels: 'AffinityFeature',
        relationshipType: 'HAS_AFFINITY_FEATURE'
    },
    {undirectedRelationshipTypes: ['HAS_AFFINITY_FEATURE']}
) AS g
RETURN g.graphName AS graph, g.nodeCount AS nodes, g.relationshipCount AS relationships;

CALL gds.nodeSimilarity.filtered.stream('customerAffinitySimilarity', {
    topK: 3,
    sourceNodeFilter: 'Customer',
    targetNodeFilter: 'Customer',
    similarityMetric: 'OVERLAP'
})
YIELD node1, node2, similarity
RETURN gds.util.asNode(node1).customerId AS Customer1,
       gds.util.asNode(node2).customerId AS Customer2,
       similarity
ORDER BY similarity DESC, Customer1, Customer2;

CALL gds.graph.drop('customerAffinitySimilarity', false);

// D. KNN - COSINE ON 14-DIMENSION BEHAVIOUR VECTOR
CALL gds.graph.drop('customerSpendKNN', false);

CALL gds.graph.project(
    'customerSpendKNN',
    'Customer',
    '*',
    {nodeProperties: ['behaviourVector']}
)
YIELD graphName, nodeCount, relationshipCount
RETURN graphName, nodeCount, relationshipCount;

CALL gds.knn.stream('customerSpendKNN', {
    topK: 3,
    nodeProperties: {behaviourVector: 'COSINE'},
    randomSeed: 42,
    concurrency: 1,
    sampleRate: 1.0,
    deltaThreshold: 0.0
})
YIELD node1, node2, similarity
RETURN gds.util.asNode(node1).customerId AS Customer1,
       gds.util.asNode(node2).customerId AS Customer2,
       similarity
ORDER BY similarity DESC, Customer1, Customer2;

CALL gds.graph.drop('customerSpendKNN', false);
